function compile_ts2rust(): void {
    let count: number = 100;

    let x: number[] = [];
    let y: number[] = [];
    let vx: number[] = [];
    let vy: number[] = [];
    let mass: number[] = [];

    let i: number = 0;
    while (i < count) {
        x.push(Math.random() * 100);
        y.push(Math.random() * 100);
        vx.push(0);
        vy.push(0);
        mass.push(Math.random() * 10 + 1);

        i = i + 1;
    }

    // step of simulation
    let step = function (dt: number): void {

        // новые массивы скоростей
        let newVx: number[] = [];
        let newVy: number[] = [];

        let i: number = 0;
        while (i < count) {

            let fx: number = 0;
            let fy: number = 0;

            let j: number = 0;
            while (j < count) {

                if (i !== j) {
                    let dx: number = x[j] - x[i];
                    let dy: number = y[j] - y[i];
                    let dist: number = Math.sqrt(dx * dx + dy * dy);

                    if (dist > 0) {
                        let f: number = (mass[i] * mass[j]) / (dist * dist);

                        fx = fx + f * (dx / dist);
                        fy = fy + f * (dy / dist);
                    }
                }

                j = j + 1;
            }


            newVx.push(vx[i] + fx * dt);
            newVy.push(vy[i] + fy * dt);

            i = i + 1;
        }

        // new positions
        let newX: number[] = [];
        let newY: number[] = [];

        let k: number = 0;
        while (k < count) {
            newX.push(x[k] + newVx[k] * dt);
            newY.push(y[k] + newVy[k] * dt);

            k = k + 1;
        }

        x = newX;
        y = newY;
        vx = newVx;
        vy = newVy;
    };

    // 100000 шагов
    let t: number = 0;
    while (t < 400000) {
        step(0.01);
        t = t + 1;
    }

    let ii: number = 0;
    while (ii < count) {
        console.log(
            "Particle " + ii +
            ": position=(" + x[ii] + ", " + y[ii] + ")," +
            " velocity=(" + vx[ii] + ", " + vy[ii] + ")"
        );
        ii = ii + 1;
    }
}

compile_ts2rust();
