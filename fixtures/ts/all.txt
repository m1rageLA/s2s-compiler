function factorial(n: number): number {
    if (n <= 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

function sumRange(n: number): number {
    let total: number = 0;
    for (let i = 1; i <= n; i = i + 1) {
        total = total + i;
    }
    return total;
}

function fibonacci(n: number): number {
    if (n <= 1) {
        return n;
    }
    return fibonacci(n - 1) + fibonacci(n - 2);
}

const double = (value: number): number => value * 2;
const increment = (value: number): number => value + 1;

const pipeline = (value: number): number => {
    const first = increment(value);
    return double(first);
};

console.log('=== math and functions ===');
const fact5 = factorial(5);
const sum5 = sumRange(5);
const fib7 = fibonacci(7);
const pipelineResult = pipeline(3);
console.log(`fact=${fact5}`);
console.log(`sum=${sum5}`);
console.log(`fib=${fib7}`);
console.log(`pipeline=${pipelineResult}`);

function labelFor(score: number): string {
    if (score > 5) {
        return 'high';
    }
    return 'low';
}

const grade = score >= 9 ? 'A' : (score >= 7 ? 'B' : 'C');
const label = labelFor(score);
const status = label === 'high' ? 'pass' : 'fail';
