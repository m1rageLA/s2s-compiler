const result = await heavy(() => {
    function go() {
        function A(i: number, j: number): number {
            return 1.0 / ((i + j) * (i + j + 1) / 2 + i + 1);
        }

        function multiplyAv(n: number, v: number[], out: number[]) {
            for (let i = 0; i < n; i++) {
                let sum: number = 0.0;
                for (let j = 0; j < n; j++) {
                    sum += A(i, j) * v[j];
                }
                out[i] = sum;
            }
        }

        function multiplyAtv(n: number, v: number[], out: number[]) {
            for (let i = 0; i < n; i++) {
                let sum: number = 0.0;
                for (let j = 0; j < n; j++) {
                    sum += A(j, i) * v[j];
                }
                out[i] = sum;
            }
        }

        function multiplyAtAv(n: number, v: number[], out: number[], tmp: number[]) {
            multiplyAv(n, v, tmp);
            multiplyAtv(n, tmp, out);
        }

        function spectralNorm(n: number): number {
            let u: number[] = [];
            let v: number[] = [];
            let tmp: number[] = [];

            for (let i = 0; i < n; i++) {
                u[i] = 1.0;
                v[i] = 0.0;
                tmp[i] = 0.0;
            }

            for (let i = 0; i < 10; i++) {
                multiplyAtAv(n, u, v, tmp);
                multiplyAtAv(n, v, u, tmp);
            }

            let vBv: number = 0.0;
            let vv: number = 0.0;

            for (let i = 0; i < n; i++) {
                vBv += u[i] * v[i];
                vv += v[i] * v[i];
            }

            return Math.sqrt(vBv / vv);
        }

        return spectralNorm(40000);
    }
    return go();

});
